import { useState } from 'react';

export const CarouselGenerator = ({ imageData, pixelsAbove, pixelsBelow }) => {
  const [activeIndex, setActiveIndex] = useState(0);
  const slideCount = imageData.length;

  const showPrevious = () => {
    setActiveIndex((current) => (current + slideCount - 1) % slideCount);
  };

  const showNext = () => {
    setActiveIndex((current) => (current + 1) % slideCount);
  };

  if (!pixelsAbove || pixelsAbove < 0) {
    pixelsAbove = 0;
  }
  if (!pixelsBelow || pixelsBelow < 0) {
    pixelsBelow = 0;
  }

  return (
    <div className="cw-carousel-stack">
      <div style={{ height: `${pixelsAbove}px` }} />
      <div className="cw-carousel">
        {slideCount > 1 && (
          <button
            type="button"
            className="cw-carousel-arrow cw-carousel-arrow--previous"
            aria-label="Previous image"
            onClick={showPrevious}
          >
            &lsaquo;
          </button>
        )}
        {imageData.map((datum, index) => {
          return (
            <img
              key={datum.src}
              className="cw-carousel-image"
              src={datum.src}
              alt={datum.alt ?? ''}
              hidden={index !== activeIndex}
            />
          );
        })}
        {slideCount > 1 && (
          <button
            type="button"
            className="cw-carousel-arrow cw-carousel-arrow--next"
            aria-label="Next image"
            onClick={showNext}
          >
            &rsaquo;
          </button>
        )}
      </div>
      {slideCount > 1 && (
        <div className="cw-carousel-dots" aria-label="Select image">
          {imageData.map((datum, index) => (
            <button
              key={datum.src}
              type="button"
              className={
                index === activeIndex
                  ? 'cw-carousel-dot cw-carousel-dot--active'
                  : 'cw-carousel-dot'
              }
              aria-label={`Show image ${index + 1}`}
              aria-current={index === activeIndex}
              onClick={() => setActiveIndex(index)}
            />
          ))}
        </div>
      )}
      <div style={{ height: `${pixelsBelow}px` }} />
    </div>
  );
};

export const TabbedCarouselGenerator = ({ tabMap, pixelsAbove, pixelsBelow }) => {
  const [activeTabKey, setActiveTabKey] = useState(Object.keys(tabMap)[0]);

  const onTabChange = (key) => {
    setActiveTabKey(key);
  };

  if (!pixelsAbove || pixelsAbove < 0) {
    pixelsAbove = 0;
  }
  if (!pixelsBelow || pixelsBelow < 0) {
    pixelsBelow = 0;
  }

  return (
    <div className="cw-carousel-stack">
      <div style={{ height: `${pixelsAbove}px` }} />
      <div className="cw-tabbed-carousel">
        <div className="cw-tabbed-carousel-tabs" role="tablist">
          {Object.entries(tabMap).map(([key, tab]) => (
            <button
              key={key}
              type="button"
              className={
                key === activeTabKey
                  ? 'cw-tabbed-carousel-tab cw-tabbed-carousel-tab--active'
                  : 'cw-tabbed-carousel-tab'
              }
              role="tab"
              aria-selected={key === activeTabKey}
              onClick={() => onTabChange(key)}
            >
              {tab.label}
            </button>
          ))}
        </div>
        <div className="cw-tabbed-carousel-panel" role="tabpanel">
          {tabMap[activeTabKey].component}
        </div>
      </div>
      <div style={{ height: `${pixelsBelow}px` }} />
    </div>
  );
};
